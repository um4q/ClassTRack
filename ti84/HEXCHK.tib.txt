# HEXCHK - hex add/subtract + DGH-style 2-character checksums
# for the TI-84 Plus CE.  Build the .8xp with:  python ti84/build.py
#
# Source format: one TI-BASIC line per text line, written with the
# calculator's own token names ("Disp ", "→", "Str1", "sub(", ...).
# Lines starting with "#" are comments and are NOT sent to the calculator.
#
# Variables: Str0 = hex digits, Str8 = ASCII lookup (see below), Str1-Str7
# and Str9 = work strings, A-Z = numbers.  All Str vars are deleted on QUIT.
#
# Rules kept everywhere:
# * Goto / Menu( only at the top level, never from inside an unfinished
#   If-Then / For( / While block (that leaks memory and ends in ERR:MEMORY).
# * Disp can't keep text on the last of the 10 rows (writing there scrolls
#   the screen), so at most 9 lines are on screen at once; every Pause is
#   preceded by a "(PRESS ENTER)" line, counted in that budget.
# * Never work on an empty string. Before every Input the string gets the
#   sentinel "@" (which can't be typed on the keypad), so a blank ENTER is
#   caught whether the OS stores "" or keeps the old value.
#
ClrHome
"0123456789ABCDEF"→Str0
# ASCII 32-126 in order, minus " (34, can't go in a TI string) and minus
# a-z (97-122, can't be typed on a CE, and their 2-byte tokens share a
# second byte with sin( cos( log( etc., which a user CAN type by mistake).
# Position C -> code C+31, +1 after the missing ", +26 after the missing a-z.
" !#$%&'()*+,-./0123456789:;<=>?@ABCDEFGHIJKLMNOPQRSTUVWXYZ[\]^_`{|}~"→Str8
#
# ---------------------------------------------------------------- main menu
Lbl M
Menu("HEX + CHECKSUM","HEX ADD/SUBTRACT",A,"DEC TO HEX",D,"DGH CHECKSUM",B,"VERIFY A CHECKSUM",V,"HEX BYTES CHECKSUM",H,"HELP",K,"QUIT",Q)
#
# ------------------------------------------------- hex add / dec to hex
# Same code for both; B = number base of what the user types (16 or 10).
Lbl A
16→B
Goto N
Lbl D
10→B
Lbl N
ClrHome
If B=16:Then
Disp "HEX ADD/SUBTRACT","EG 24+31+52+44","A-F: PRESS ALPHA FIRST"
Else
Disp "DECIMAL TO HEX","EG 235  OR  200+35"
End
Disp "BLANK ENTRY = MENU"
Lbl N2
"@"→Str1
Input ">",Str1
If length(Str1)=0:Goto M
If Str1="@":Goto M
# T = total, N = number being typed (K = has digits), S = its sign.
# + space , end a number; - and (-) flip the sign, so 10-⁻5 = 15.
# N stops growing at 1E12 (flagged by O) so 90 F's can't hit ERR:OVERFLOW.
# Each number and the answer must be < 1E12; the running total only has to
# stay < 1E13 so it stays exact (14 digits) along the way.
0→T
0→N
0→K
1→S
0→P
0→O
For(I,1,length(Str1))
sub(Str1,I,1)→Str2
inString(sub(Str0,1,B),Str2)→X
If X:Then
If N<1ᴇ12:B*N+X-1→N
If N≥1ᴇ12:1→O
1→K
Else
If Str2="+" or Str2="-" or Str2="⁻" or Str2=" " or Str2=",":Then
T+S*N→T
If abs(T)≥1ᴇ13:1→O
If K:1→S
0→N
0→K
If Str2="-" or Str2="⁻":⁻S→S
Else
If not(P):I→P
End
End
End
T+S*N→T
If abs(T)≥1ᴇ12:1→O
If P:Then
Disp "NOT A DIGIT:",sub(Str1,P,1)
End
If O and not(P):Then
Disp "NUMBER TOO BIG, MAX IS"
If B=16:Disp "E8D4A50FFF HEX"
If B=10:Disp "999999999999"
End
If P or O:Goto N2
# hex of |T| -> Str6, decimal of |T| -> Str5
abs(T)→V
sub(Str0,V-16*int(V/16)+1,1)→Str6
int(V/16)→V
While V
sub(Str0,V-16*int(V/16)+1,1)+Str6→Str6
int(V/16)→V
End
abs(T)→V
sub(Str0,V-10*int(V/10)+1,1)→Str5
int(V/10)→V
While V
sub(Str0,V-10*int(V/10)+1,1)+Str5→Str5
int(V/10)→V
End
If T<0:Then
"-"+Str6→Str6
"-"+Str5→Str5
End
# low byte (T mod 256, two's complement when T<0)
T-256*int(T/256)→L
sub(Str0,int(L/16)+1,1)+sub(Str0,L-16*int(L/16)+1,1)→Str3
Disp "= "+Str6+" HEX","= "+Str5+" DEC"
If T≥0:Then
Disp "LAST 2 HEX DIGITS: "+Str3
Else
Disp "8-BIT 2'S COMP: "+Str3
End
Goto N2
#
# -------------------------------------- DGH checksum / verify a checksum
# W=0: work out the checksum.  W=1: check one typed on the end.
Lbl B
0→W
Goto F
Lbl V
1→W
Lbl F
Menu("FIRST CHARACTER","$  COMMAND (SHORT)",F1,"#  COMMAND (LONG)",F2,"*  REPLY",F3,"?  ERROR REPLY",F4,"NONE / TYPE IT ALL",F5,"BACK",M)
# Str7 = the first character; it is also the Input prompt, so the user
# sees e.g. $1RD while typing 1RD. ">" means "no first character".
Lbl F1
"$"→Str7
Goto F7
Lbl F2
"#"→Str7
Lbl F7
ClrHome
Disp "TYPE ADDRESS + COMMAND","(+ DATA IF ANY), EG 1RD"
Goto F6
Lbl F3
"*"→Str7
ClrHome
Disp "TYPE THE REST OF THE REPLY","EG 1RD+00072.10"
Goto F6
Lbl F4
"?"→Str7
ClrHome
Disp "TYPE THE REST OF THE REPLY"
If W=0:Disp "EG 1 BAD CHECKSUM"
Goto F6
Lbl F5
">"→Str7
ClrHome
Disp "TYPE THE WHOLE STRING"
Lbl F6
If W=1:Disp "THEN ITS 2 CHECKSUM CHARS"
"@"→Str1
Input Str7,Str1
If length(Str1)=0:Goto F
If Str1="@":Goto F
If Str7≠">":Str7+Str1→Str1
If length(Str1)>120:Then
Disp "TOO LONG (MAX 120)","(PRESS ENTER)"
Pause
End
If length(Str1)>120:Goto F
If W=1 and length(Str1)<3:Then
Disp "TOO SHORT - TYPE THE","STRING AND ITS 2-CHARACTER","CHECKSUM","(PRESS ENTER)"
Pause
End
If W=1 and length(Str1)<3:Goto F
If W=1:Then
sub(Str1,length(Str1)-1,2)→Str9
sub(Str1,1,length(Str1)-2)→Str1
End
# Show every character as c=XX (its ASCII code in hex), 5 to a line, 8
# lines a page, and add them up in S. Str4 = line being built (R pairs in
# it), Q = lines on this page. Str6 collects the characters as counted
# ((-) becomes -), behind a "?" so it is never empty.
ClrHome
0→S
0→P
0→R
0→Q
"?"→Str6
For(I,1,length(Str1))
sub(Str1,I,1)→Str2
If Str2="⁻":"-"→Str2
Str6+Str2→Str6
inString(Str8,Str2)→C
If C:C+31+(C>2)+26*(C>64)→C
If not(C) and not(P):I→P
S+C→S
If C:Then
sub(Str0,int(C/16)+1,1)+sub(Str0,C-16*int(C/16)+1,1)→Str3
Else
"?"→Str2
"??"→Str3
End
If R:Then
Str4+" "+Str2+"="+Str3→Str4
Else
Str2+"="+Str3→Str4
End
R+1→R
If R=5:Then
Disp Str4
0→R
Q+1→Q
End
If Q=8 and I<length(Str1):Then
Disp "(PRESS ENTER)"
Pause
ClrHome
0→Q
End
End
If R:Then
Disp Str4
Q+1→Q
End
# G=1: the result fits after "SEND THIS: " on one line.
# E = lines still to come, "(PRESS ENTER)" included: the error (3); or
# SUM, CHECKSUM, then 2 verify lines, or the result (1 line, or a label +
# 26-column pieces). If they don't fit under the Q lines in the 9 usable
# rows, wait and start a new screen.
(Str7="$" or Str7="#") and length(Str1)<14→G
If P:3→E
If not(P) and W=1:5→E
If not(P) and W=0:4+not(G)*int((length(Str1)+27)/26)→E
If Q+E>9:Then
Disp "(PRESS ENTER)"
Pause
ClrHome
End
If P:Then
Disp "CAN'T USE THIS CHARACTER:",sub(Str1,P,1),"(PRESS ENTER)"
Pause
End
If P:Goto F
# hex of S -> Str4
S→V
sub(Str0,V-16*int(V/16)+1,1)→Str4
int(V/16)→V
While V
sub(Str0,V-16*int(V/16)+1,1)+Str4→Str4
int(V/16)→V
End
# Str4 = sum in hex, Str5 = sum in decimal, Str3 = checksum (last 2 hex)
S→V
sub(Str0,V-10*int(V/10)+1,1)→Str5
int(V/10)→V
While V
sub(Str0,V-10*int(V/10)+1,1)+Str5→Str5
int(V/10)→V
End
S-256*int(S/256)→L
sub(Str0,int(L/16)+1,1)+sub(Str0,L-16*int(L/16)+1,1)→Str3
Disp "SUM = "+Str4+" HEX ("+Str5+" DEC)","CHECKSUM = "+Str3
If W=0:Then
sub(Str6,2,length(Str1))+Str3→Str5
If G:Disp "SEND THIS: "+Str5
If not(G):Then
If Str7="$" or Str7="#":Disp "SEND THIS:"
If Str7≠"$" and Str7≠"#":Disp "WITH CHECKSUM:"
For(I,1,length(Str5),26)
Disp sub(Str5,I,min(26,length(Str5)-I+1))
End
End
Else
If Str9=Str3:Then
Disp "RECEIVED "+Str3,"MATCH - CHECKSUM OK"
Else
Disp "WRONG! RECEIVED:",Str9
End
End
Disp "(PRESS ENTER)"
Pause
Goto F
#
# ------------------------------------------------- hex bytes checksum
# Byte pairs, e.g. Intel HEX "0300300002337A" or a Modbus ASCII frame.
Lbl H
ClrHome
Disp "HEX BYTES CHECKSUM","TYPE BYTES (2 DIGITS EACH)","EG 0300300002337A","SPACES ARE IGNORED","BLANK ENTRY = MENU"
Lbl H2
"@"→Str1
Input "BYTES:",Str1
If length(Str1)=0:Goto M
If Str1="@":Goto M
0→S
0→X
0→N
0→K
0→J
0→P
For(I,1,length(Str1))
sub(Str1,I,1)→Str2
inString(Str0,Str2)→D
If D:Then
16*N+D-1→N
K+1→K
If K=2:Then
S+N→S
J+1→J
# X = X xor N, one bit at a time
0→Y
For(Z,0,7)
Y+2^Z*(int(X/2^Z)+int(N/2^Z)-2*int((int(X/2^Z)+int(N/2^Z))/2))→Y
End
Y→X
0→N
0→K
End
Else
If Str2≠" " and Str2≠":" and Str2≠"," and not(P):I→P
End
End
If P:Then
Disp "NOT A HEX DIGIT:",sub(Str1,P,1)
End
If K=1 and not(P):Then
Disp "ODD NUMBER OF DIGITS"
End
If P or K=1:Goto H2
S→V
sub(Str0,V-16*int(V/16)+1,1)→Str4
int(V/16)→V
While V
sub(Str0,V-16*int(V/16)+1,1)+Str4→Str4
int(V/16)→V
End
J→V
sub(Str0,V-10*int(V/10)+1,1)→Str5
int(V/10)→V
While V
sub(Str0,V-10*int(V/10)+1,1)+Str5→Str5
int(V/10)→V
End
S-256*int(S/256)→L
sub(Str0,int(L/16)+1,1)+sub(Str0,L-16*int(L/16)+1,1)→Str3
256-L-256*(L=0)→L
sub(Str0,int(L/16)+1,1)+sub(Str0,L-16*int(L/16)+1,1)→Str6
sub(Str0,int(X/16)+1,1)+sub(Str0,X-16*int(X/16)+1,1)→Str7
Disp "BYTES: "+Str5,"SUM = "+Str4+" HEX","SUM, LAST 2 DIGITS: "+Str3,"2'S COMP (INTEL/LRC): "+Str6,"XOR OF BYTES: "+Str7
Goto H2
#
# ---------------------------------------------------------------- help
Lbl K
ClrHome
Disp "DGH CHECKSUM = LAST 2 HEX","DIGITS OF THE SUM OF THE","ASCII CODES OF EVERY CHAR:","PROMPT ($ OR #), ADDRESS,","COMMAND, DATA (NOT THE CR)","EG $1RD:","24+31+52+44 = EB","SEND $1RDEB","(PRESS ENTER)"
Pause
ClrHome
Disp "$ GETS A SHORT REPLY WITH","NO CHECKSUM. # GETS A LONG","REPLY THAT ECHOES THE","COMMAND AND ENDS IN A","CHECKSUM:","#1RD -> *1RD+00072.10A4","CHECK IT: VERIFY, THEN *","NO SPACES IN COMMANDS","(PRESS ENTER)"
Pause
ClrHome
Disp "KEYS:","ALPHA = ONE LETTER","2ND ALPHA = LETTER LOCK","* = THE MULTIPLY KEY","- = MINUS OR (-) KEY","$ # ARE IN THE","FIRST-CHARACTER MENU","BLANK ENTRY = GO BACK","(PRESS ENTER)"
Pause
Goto M
#
# ---------------------------------------------------------------- quit
Lbl Q
DelVar Str0
DelVar Str1
DelVar Str2
DelVar Str3
DelVar Str4
DelVar Str5
DelVar Str6
DelVar Str7
DelVar Str8
DelVar Str9
ClrHome
# a value on the last line replaces the "Done" message
"
